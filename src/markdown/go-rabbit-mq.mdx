RabbitMQ is an essential tool in the microservices world, enabling scalable, reliable, and decoupled communication between services. In this guide, we'll walk through practical examples of using RabbitMQ with Go, including installation, exchange types, queues, and handling failures with dead-letter queues.

## Installing RabbitMQ with Docker Compose

Create a `docker-compose.yml` file:

```yaml
version: "3"
services:
  rabbitmq:
    image: rabbitmq:4.0-management
    ports:
      - "5672:5672"
      - "15672:15672"
    environment:
      RABBITMQ_DEFAULT_USER: guest
      RABBITMQ_DEFAULT_PASS: guest
```

Start RabbitMQ:

```sh
docker-compose up -d
```

Access the management UI at [http://localhost:15672](http://localhost:15672) (user: guest, pass: guest).

## Two Main Concepts in RabbitMQ

When starting with RabbitMQ, there are two main concepts you should know:

### 1. Queue

A **Queue** is a buffer that stores messages until they are processed by consumers. It ensures that messages are delivered reliably and can be consumed at a later time.

### 2. Exchange

An **Exchange** in RabbitMQ is a routing mechanism that receives messages from producers and routes them to queues based on rules called bindings. There are several types of exchanges.

Let's start with the simplest type of exchange.

### a. Direct Exchange

A **Direct Exchange** delivers messages to queues based on a message routing key that exactly matches.

**For the first example, I'll show you the full code for RabbitMQ with Go. In the next examples, the code will be shorter.**

```go
package main

import (
	"context"
	"fmt"
	"log"
	"time"

	amqp "github.com/rabbitmq/amqp091-go"
)

func handleError(err error, msg string) {
	if err != nil {
		log.Fatalf("%s: %s", msg, err)
	}
}

func main() {
	conn, err := amqp.Dial("amqp://guest:guest@localhost:5672/")
	handleError(err, "Failed to connect to RabbitMQ")
	defer conn.Close()

	ch, err := conn.Channel()
	handleError(err, "Failed to open a channel")
	defer ch.Close()

	exchangeName := "my_direct_exchange"
	err = ch.ExchangeDeclare(
		exchangeName, // name
		"direct",     // type
		true,         // durable
		false,        // auto-deleted
		false,        // internal
		false,        // no-wait
		nil,          // arguments
	)
	handleError(err, "Failed to declare an exchange")
	fmt.Printf("Declared Direct Exchange: %s\n", exchangeName)

	queueName1 := "queue_A"
	q1, err := ch.QueueDeclare(
		queueName1, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive (only this connection can use it)
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue A")
	fmt.Printf("Declared Queue A: %s\n", q1.Name)

	queueName2 := "queue_B"
	q2, err := ch.QueueDeclare(
		queueName2, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue B")
	fmt.Printf("Declared Queue B: %s\n", q2.Name)

	keyA := "keyA"
	keyB := "keyB"

	err = ch.QueueBind(
		q1.Name,           // queue name
		keyA,              // binding key for queue A
		exchangeName,      // exchange
		false,             // no-wait
		nil,               // arguments
	)
	handleError(err, fmt.Sprintf("Failed to bind queue %s with key '%s'", q1.Name, keyA))
	fmt.Printf("Bound Queue %s to %s with binding key '%s'\n", q1.Name, exchangeName, keyA)

	err = ch.QueueBind(
		q2.Name,           // queue name
		keyB,              // binding key for queue B
		exchangeName,      // exchange
		false,             // no-wait
		nil,               // arguments
	)
	handleError(err, fmt.Sprintf("Failed to bind queue %s with key '%s'", q2.Name, keyB))
	fmt.Printf("Bound Queue %s to %s with binding key '%s'\n", q2.Name, exchangeName, keyB)

	fmt.Println("\n--- Starting Consumers ---")

	go consumeMessages(ch, q1.Name, "Consumer A")
	go consumeMessages(ch, q2.Name, "Consumer B")

	time.Sleep(1 * time.Second)

	fmt.Println("\n--- Publishing Messages ---")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	publishList := []struct {
		key   string
		body  string
	}{
		{keyA, "Hello to A"},
		{keyB, "Hello to B"},
		{keyA, "Another for A"},
		{keyB, "Another for B"},
	}

	for i, msg := range publishList {
		err = ch.PublishWithContext(
			ctx,
			exchangeName, // exchange
			msg.key,      // routing key
			false,        // mandatory
			false,        // immediate
			amqp.Publishing{
				ContentType: "text/plain",
				Body:        []byte(fmt.Sprintf("Message %d: %s", i+1, msg.body)),
			})
		handleError(err, "Failed to publish a message")
		fmt.Printf("Published message to %s with routing key '%s': %s\n", exchangeName, msg.key, msg.body)
		time.Sleep(50 * time.Millisecond)
	}

	fmt.Println("\n--- All messages published. Waiting for consumers... ---")
	select {}
}

func consumeMessages(ch *amqp.Channel, queueName string, consumerName string) {
	msgs, err := ch.Consume(
		queueName,    // queue
		consumerName, // consumer
		true,         // auto-ack
		false,        // exclusive
		false,        // no-local
		false,        // no-wait
		nil,          // args
	)
	handleError(err, fmt.Sprintf("%s: Failed to register a consumer", consumerName))

	fmt.Printf("%s: Waiting for messages on queue '%s'...\n", consumerName, queueName)

	for d := range msgs {
		fmt.Printf("%s: Received a message: %s\n", consumerName, d.Body)
	}
	fmt.Printf("%s: Consumer stopped.\n", consumerName)
}
```

### b. Fanout Exchange

A **Fanout Exchange** routes messages to all queues that are bound to it, regardless of the routing key.

```go
		true,         // durable
		false,        // auto-deleted
		false,        // internal
		false,        // no-wait
		nil,          // arguments
	)
	handleError(err, "Failed to declare an exchange")
	fmt.Printf("Declared Fanout Exchange: %s\n", exchangeName)

	queueName1 := "queue_A"
	q1, err := ch.QueueDeclare(
		queueName1, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive (only this connection can use it)
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue A")
	fmt.Printf("Declared Queue A: %s\n", q1.Name)

	queueName2 := "queue_B"
	q2, err := ch.QueueDeclare(
		queueName2, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue B")
	fmt.Printf("Declared Queue B: %s\n", q2.Name)

	emptyKey := ""

		nil,               // arguments
	)
	handleError(err, fmt.Sprintf("Failed to bind queue %s", q1.Name))
	fmt.Printf("Bound Queue %s to %s\n", q1.Name, exchangeName)

	err = ch.QueueBind(
		q2.Name,           // queue name
		emptyKey,          // binding key (ignored for fanout)

		false,             // no-wait

	handleError(err, fmt.Sprintf("Failed to bind queue %s", q2.Name))

	go consumeMessages(ch, q1.Name, "Consumer A")

				ContentType: "text/plain",
				Body:        []byte(fmt.Sprintf("Message %d: %s", i+1, body)),

		handleError(err, "Failed to publish a message")
		fmt.Printf("Published message to %s: %s\n", exchangeName, body)
		time.Sleep(50 * time.Millisecond)
```

### c. Topic Exchange

A **Topic Exchange** routes messages to one or more queues based on matching between a message's routing key and the pattern used in the binding.

In this example, we declare a topic exchange and bind two queues with different patterns: `queue_A` is bound with the pattern `foo.*` (matches any routing key starting with 'foo.' and one more word), and `queue_B` is bound with the pattern `foo.bar` (matches only the exact routing key 'foo.bar'). When publishing messages with routing keys like 'foo.bar', 'foo.baz', and 'foo.qux', you will see that:

```go
func main() {
	// Initial of RabbitMQ is the same code of above
	...

	exchangeName := "my_topic_exchange"
	err = ch.ExchangeDeclare(
		exchangeName, // name
		"topic",     // type
		true,         // durable
		false,        // auto-deleted
		false,        // internal
		false,        // no-wait
		nil,          // arguments
	)
	handleError(err, "Failed to declare an exchange")
	fmt.Printf("Declared Topic Exchange: %s\n", exchangeName)

	queueName1 := "queue_A"
	q1, err := ch.QueueDeclare(
		queueName1, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive (only this connection can use it)
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue A")
	fmt.Printf("Declared Queue A: %s\n", q1.Name)

	queueName2 := "queue_B"
	q2, err := ch.QueueDeclare(
		queueName2, // name
		false,      // durable
		false,      // delete when unused
		false,      // exclusive
		false,      // no-wait
		nil,        // arguments
	)
	handleError(err, "Failed to declare queue B")
	fmt.Printf("Declared Queue B: %s\n", q2.Name)

	patternA := "foo.*"
	patternB := "foo.bar"

	err = ch.QueueBind(
		q1.Name,           // queue name
		patternA,          // binding key (topic pattern)
		exchangeName,      // exchange
		false,             // no-wait
		nil,               // arguments
	)
	handleError(err, fmt.Sprintf("Failed to bind queue %s with pattern '%s'", q1.Name, patternA))
	fmt.Printf("Bound Queue %s to %s with pattern '%s'\n", q1.Name, exchangeName, patternA)

	err = ch.QueueBind(
		q2.Name,           // queue name
		patternB,          // binding key (topic pattern)
		exchangeName,      // exchange
		false,             // no-wait
		nil,               // arguments
	)
	handleError(err, fmt.Sprintf("Failed to bind queue %s with pattern '%s'", q2.Name, patternB))
	fmt.Printf("Bound Queue %s to %s with pattern '%s'\n", q2.Name, exchangeName, patternB)

	fmt.Println("\n--- Starting Consumers ---")

	go consumeMessages(ch, q1.Name, "Consumer A")
	go consumeMessages(ch, q2.Name, "Consumer B")

	time.Sleep(1 * time.Second)

	fmt.Println("\n--- Publishing Messages ---")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	publishList := []struct {
		key  string
		body string
	}{
		{"foo.bar", "Message for foo.bar"},
		{"foo.baz", "Message for foo.baz"},
		{"foo.qux", "Message for foo.qux"},
		{"bar.foo", "Message for bar.foo (should not match)"},
	}

	for i, msg := range publishList {
		err = ch.PublishWithContext(
			ctx,
			exchangeName, // exchange
			msg.key,      // routing key
			false,        // mandatory
			false,        // immediate
			amqp.Publishing{
				ContentType: "text/plain",
				Body:        []byte(fmt.Sprintf("Message %d: %s", i+1, msg.body)),
			})
		handleError(err, "Failed to publish a message")
		fmt.Printf("Published message to %s with routing key '%s': %s\n", exchangeName, msg.key, msg.body)
		time.Sleep(50 * time.Millisecond)
	}

    ...
}
```

## Bonus: Handling Failures with Dead Letter Queue (DLQ)

A **Dead Letter Exchange (DLX)** in RabbitMQ is used to handle messages that cannot be processed successfully, such as those that are rejected, expire, or cannot be routed. When a message is dead-lettered, RabbitMQ automatically routes it to the DLX, which then delivers it to a designated dead-letter queue (DLQ). This allows you to inspect, retry, or log failed messages separately from your main processing flow.

For example, imagine a payment service that processes transactions. If a transaction fails validation, the message can be sent to a dead-letter queue for later inspection or retry, ensuring no data is lost and failures are handled gracefully.

```go
func main() {
	// Initial of RabbitMQ is the same code of above
	...

	dlxName := "my_dlx"
	dlqName := "my_dlq"

	err = ch.ExchangeDeclare(
		dlxName,   // name
		"fanout", // type (DLX is often fanout or direct)
		true,      // durable
		false,     // auto-deleted
		false,     // internal
		false,     // no-wait
		nil,       // arguments
	)
	handleError(err, "Failed to declare DLX")
	fmt.Printf("Declared Dead Letter Exchange: %s\n", dlxName)

	dlq, err := ch.QueueDeclare(
		dlqName, // name
		false,   // durable
		false,   // delete when unused
		false,   // exclusive
		false,   // no-wait
		nil,     // arguments
	)
	handleError(err, "Failed to declare DLQ")
	fmt.Printf("Declared Dead Letter Queue: %s\n", dlq.Name)

	err = ch.QueueBind(
		dlq.Name, // queue name
		"",       // routing key (fanout ignores this)
		dlxName,  // exchange
		false,    // no-wait
		nil,      // arguments
	)
	handleError(err, "Failed to bind DLQ to DLX")
	fmt.Printf("Bound DLQ %s to DLX %s\n", dlq.Name, dlxName)

	mainQueueName := "main_queue"
	mainQueueArgs := amqp.Table{
		"x-dead-letter-exchange": dlxName,
	}
	mainQ, err := ch.QueueDeclare(
		mainQueueName, // name
		false,         // durable
		false,         // delete when unused
		false,         // exclusive
		false,         // no-wait
		mainQueueArgs, // arguments
	)
	handleError(err, "Failed to declare main queue with DLX")
	fmt.Printf("Declared Main Queue: %s (with DLX: %s)\n", mainQ.Name, dlxName)

	fmt.Println("\n--- Starting Consumers ---")
	go consumeAndReject(ch, mainQ.Name, "Main Consumer")
	go consumeMessages(ch, dlq.Name, "DLQ Consumer")

	time.Sleep(1 * time.Second)

	fmt.Println("\n--- Publishing Message to Main Queue ---")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	err = ch.PublishWithContext(
		ctx,
		"",            // default exchange
		mainQ.Name,     // routing key (queue name)
		false,          // mandatory
		false,          // immediate
		amqp.Publishing{
			ContentType: "text/plain",
			Body:        []byte("This message will be dead-lettered!"),
		},
	)

    ...
}

func consumeAndReject(ch *amqp.Channel, queueName string, consumerName string) {
	msgs, err := ch.Consume(
		queueName,    // queue
		consumerName, // consumer
		false,        // auto-ack (set to false to allow rejection)
		false,        // exclusive
		false,        // no-local
		false,        // no-wait
		nil,          // args
	)
	handleError(err, fmt.Sprintf("%s: Failed to register a consumer", consumerName))

	fmt.Printf("%s: Waiting for messages on queue '%s'...\n", consumerName, queueName)

	for d := range msgs {
		fmt.Printf("%s: Received a message: %s\n", consumerName, d.Body)
		// Simulate processing failure and reject the message (dead-letter it)
		d.Reject(false) // false = do not requeue, send to DLQ
		fmt.Printf("%s: Rejected message, sent to DLQ.\n", consumerName)
	}
	fmt.Printf("%s: Consumer stopped.\n", consumerName)
}
```
